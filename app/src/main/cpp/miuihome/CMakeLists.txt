cmake_minimum_required(VERSION 3.22)
project(myinjector_miuihome LANGUAGES C CXX ASM)
add_library(miuihome_xz STATIC xz/xz_crc32.c xz/xz_crc64.c xz/xz_dec_lzma2.c xz/xz_dec_stream.c)
target_compile_definitions(miuihome_xz PRIVATE XZ_DEC_SINGLE XZ_USE_CRC64)
target_compile_options(miuihome_xz PRIVATE -fvisibility=hidden -ffunction-sections -fdata-sections)
set_target_properties(miuihome_xz PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(miuihome_resolver STATIC resolver_elf.cpp runtime_profile.cpp)
target_compile_features(miuihome_resolver PUBLIC cxx_std_17)
target_compile_definitions(miuihome_resolver PRIVATE XZ_USE_CRC64)
target_compile_options(miuihome_resolver PRIVATE -Wall -Wextra -Werror -fvisibility=hidden
    -fno-exceptions -fno-rtti -ffunction-sections -fdata-sections)
set_target_properties(miuihome_resolver PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(miuihome_resolver PUBLIC miuihome_xz ${CMAKE_DL_LIBS})
if(NOT ANDROID)
    add_executable(miuihome_resolver_cli ../../../../../tests/miuihome_resolver_cli.cpp)
    target_include_directories(miuihome_resolver_cli PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(miuihome_resolver_cli PRIVATE miuihome_resolver)
    return()
endif()
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The HYOS entry requires Android arm64-v8a")
endif()
add_library(myinjector_miuihome SHARED native_hook.cpp native_broadcast.cpp dart_hooks.S lsposed_hook_backend.cpp)
target_compile_features(myinjector_miuihome PRIVATE cxx_std_17)
target_compile_options(myinjector_miuihome PRIVATE
    -Wall -Wextra -Werror -fvisibility=hidden -fno-exceptions -fno-rtti
    -fno-threadsafe-statics -ffunction-sections -fdata-sections -mbranch-protection=standard)
target_link_options(myinjector_miuihome PRIVATE
    "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
    -Wl,--gc-sections -Wl,--no-undefined -Wl,-z,max-page-size=16384 -Wl,-z,relro -Wl,-z,now)
target_compile_options(miuihome_resolver PRIVATE -mbranch-protection=standard)
target_compile_options(miuihome_xz PRIVATE -mbranch-protection=standard)
target_link_libraries(myinjector_miuihome PRIVATE log dl miuihome_resolver)
